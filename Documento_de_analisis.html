Hay algunas variables que tienen efecto suelo o techo. (categoria menor o mayor <5% de respuesta). Se opto por recodificar a la categoria mas proximas en estos casos a modo de poder capturar de mejor forma la variable conntinua latente en la correlacion.

fusionar_extremos <- function(x, umbral = 0.05) {
  
  if (!is.numeric(x)) return(x)
  
  x_no_na <- x[!is.na(x)]
  prop <- prop.table(table(x_no_na))
  
  # Copia
  x_mod <- x
  
  # Efecto suelo: 1 < 5%
  if ("1" %in% names(prop) && prop["1"] < umbral) {
    x_mod[x_mod == 1] <- 2
  }
  
  # Efecto techo: 5 < 5%
  if ("5" %in% names(prop) && prop["5"] < umbral) {
    x_mod[x_mod == 5] <- 4
  }
  
  return(x_mod)
}


moduloc_fusionado <- as.data.frame(
  lapply(primera_mitad, fusionar_extremos, umbral = 0.05)
